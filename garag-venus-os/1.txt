# ── 1. Check USB serial device is present ───────────────────
ls -l /dev/ttyUSB* /dev/ttyACM* 2>/dev/null
# JBD USB-UART adapter shows as ttyACM0


# ── 2. Install the driver ───────────────────────────────────
# ⚠ CORRECT URL — path changed in recent repo restructure:
#   dbus-serialbattery/install.sh   ← correct
#   etc/dbus-serialbattery/...      ← OLD, returns 404, do not use
wget -O /tmp/install.sh https://raw.githubusercontent.com/mr-manuel/venus-os_dbus-serialbattery/master/dbus-serialbattery/install.sh
bash /tmp/install.sh
# → interactive menu: press 1 (latest stable release) → Enter
# → wait for download + install
reboot


# ── 3. After reboot: run enable.sh BEFORE anything else ─────
# ⚠ If you skip this, the service run-script is never created and
#   you get: sed: /var/volatile/services/dbus-serialbattery.ttyACM0/run: No such file or directory
#   The service will appear "up" but restart every second.
bash /data/apps/dbus-serialbattery/enable.sh
reboot


# ── 4. Write config ─────────────────────────────────────────
nano /data/apps/dbus-serialbattery/config.ini
# Delete all existing content and paste the block below exactly.
# Config must start with [DEFAULT] on line 1.

# ┌─ config.ini ──────────────────────────────────────────────
[DEFAULT]
MAX_BATTERY_CHARGE_CURRENT = 60.0
MAX_BATTERY_DISCHARGE_CURRENT = 60.0

; NMC cell voltages (14S Nissan Leaf modules)
MAX_CELL_VOLTAGE = 4.10
MIN_CELL_VOLTAGE = 3.25
FLOAT_CELL_VOLTAGE = 4.07

; Taper charge/discharge current as cells approach limits
CCCM_CV_ENABLE = True
DCCM_CV_ENABLE = True
CELL_VOLTAGES_WHILE_CHARGING = 4.10, 4.05, 4.00, 3.90
MAX_CHARGE_CURRENT_CV_FRACTION = 0, 0.05, 0.5, 1
CELL_VOLTAGES_WHILE_DISCHARGING = 3.25, 3.30, 3.35, 3.45
MAX_DISCHARGE_CURRENT_CV_FRACTION = 0, 0.1, 0.5, 1

; Display
BATTERY_CELL_DATA_FORMAT = 1
GUI_PARAMETERS_SHOW_ADDITIONAL_INFO = True

; SOC from JBD internal coulomb counter — driver does not recalculate
SOC_CALCULATION = False
SOC_RESET_CELL_VOLTAGE = 3.9
# └───────────────────────────────────────────────────────────

# Strip Windows CRLF if file was edited on Windows
sed -i 's/\r$//' /data/apps/dbus-serialbattery/config.ini

# Check for duplicate keys — output must be exactly 1 line
grep -ni '^SOC_CALCULATION' /data/apps/dbus-serialbattery/config.ini

# Apply config
/data/apps/dbus-serialbattery/restart.sh


# ════════════════════════════════════════════════════════════
